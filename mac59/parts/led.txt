/******************************************************************************
 CLedDisplay

	Twelve seven-segment cells, cell 11 leftmost. A cell is what the last
	display interval drove: its segments and its decimal point.

 ******************************************************************************/

CLedDisplay::CLedDisplay(CView *anEnclosure, CBureaucrat *aSupervisor,
						 short aWidth, short aHeight, short aHEncl, short aVEncl)
	: CPane(anEnclosure, aSupervisor, aWidth, aHeight, aHEncl, aVEncl,
			sizFIXEDLEFT, sizFIXEDTOP)
{
	short i;
	for (i = 0; i < 12; i++) { seg[i] = 0; dp[i] = 0; }
}


void CLedDisplay::SetCells(const unsigned char *aSeg, const unsigned char *aDp)
{
	short	i;
	Boolean	changed = FALSE;
	Rect	r;

	for (i = 0; i < 12; i++)
		if (seg[i] != aSeg[i] || dp[i] != aDp[i]) { seg[i] = aSeg[i]; dp[i] = aDp[i]; changed = TRUE; }
	if (changed) {
		Prepare();
		FrameToQDR(&frame, &r);
		Draw(&r);
	}
}


void CLedDisplay::GetText(char *out)
{
	short	i, d, n = 0;

	for (i = 11; i >= 0; i--) {
		if (seg[i] == kSegMinus) out[n++] = '-';
		else if (seg[i] == kSegDegree) out[n++] = 'o';
		else if (seg[i] == kSegC) out[n++] = 'C';
		else if (seg[i] != 0) {
			for (d = 0; d < 10 && kDigitSeg[d] != seg[i]; d++) ;
			out[n++] = (char) (d < 10 ? '0' + d : '?');
		}
		if (dp[i]) out[n++] = ',';
	}
	out[n] = '\0';
}


void CLedDisplay::Draw(Rect *area)
{
	Rect	r, s;
	short	i, x, y;
	unsigned char m;

	FrameToQDR(&frame, &r);
	RGBForeColor(&kLedBg);
	PaintRect(&r);
	RGBForeColor(&kRed);
	y = (short) (r.top + 8);
	for (i = 0; i < 12; i++) {
		x = (short) (r.right - 22 - 21 * i);			/* cell i, from the right	*/
		m = seg[i];
		if (m & 0x01) { SetRect(&s, x + 1, y,      x + 11, y + 2);  PaintRect(&s); }
		if (m & 0x02) { SetRect(&s, x + 10, y + 1, x + 12, y + 11); PaintRect(&s); }
		if (m & 0x04) { SetRect(&s, x + 10, y + 12, x + 12, y + 22); PaintRect(&s); }
		if (m & 0x08) { SetRect(&s, x + 1, y + 21, x + 11, y + 23); PaintRect(&s); }
		if (m & 0x10) { SetRect(&s, x,     y + 12, x + 2,  y + 22); PaintRect(&s); }
		if (m & 0x20) { SetRect(&s, x,     y + 1,  x + 2,  y + 11); PaintRect(&s); }
		if (m & 0x40) { SetRect(&s, x + 1, y + 10, x + 11, y + 12); PaintRect(&s); }
		if (dp[i])    { SetRect(&s, x + 14, y + 20, x + 17, y + 23); PaintRect(&s); }
	}
	RGBForeColor(&kBlackC);
	RGBBackColor(&kWhiteC);
}
