
/******************************************************************************
 CBodyPane

 ******************************************************************************/

CBodyPane::CBodyPane(CView *anEnclosure, CBureaucrat *aSupervisor, short aWidth, short aHeight)
	: CPane(anEnclosure, aSupervisor, aWidth, aHeight, 0, 0, sizFIXEDLEFT, sizFIXEDTOP)
{
	short i;

	info[0] = 0;
	cardName[0] = 0;
	for (i = 0; i < 5; i++) cue[i][0] = 0;
	SetWantsClicks(TRUE);
}


/* The cue card above A-E: what the module program's user keys do. */
void CBodyPane::SetCue(short i, const char *aLabel)
{
	Rect r;
	short n = 0;

	if (i < 0 || i > 4) return;
	while (aLabel[n] != '\0' && n < 31) { cue[i][n + 1] = (unsigned char) aLabel[n]; n++; }
	cue[i][0] = (unsigned char) n;
	Prepare();
	SetRect(&r, 6, 92, 290, 110);
	RefreshRect(&r);
}


/* The card's name in the strip, as the label on a card in the slot. */
void CBodyPane::SetCardName(Str255 aName)
{
	Rect r;

	BlockMove(aName, cardName, (long) aName[0] + 1);
	Prepare();
	SetRect(&r, 6, 74, 290, 92);
	RefreshRect(&r);
}


/* A click on the card strip reads a card. */
void CBodyPane::DoClick(Point hitPt, short modifierKeys, long when)
{
	if (hitPt.v >= 74 && hitPt.v < 110 && hitPt.h >= 6 && hitPt.h < 290 && itsSupervisor != NULL)
		itsSupervisor->DoCommand(cmdReadCard);
}


void CBodyPane::SetInfo(Str255 anInfo)
{
	Rect r;

	BlockMove(anInfo, info, (long) anInfo[0] + 1);
	Prepare();
	SetRect(&r, 10, 526, 170, 538);
	RefreshRect(&r);
}


void CBodyPane::Draw(Rect *area)
{
	Rect	r;
	short	x, w;

	FrameToQDR(&frame, &r);
	RGBForeColor(&kCase);
	PaintRect(&r);

	SetRect(&r, 6, 6, 290, 70);						/* the display window			*/
	RGBForeColor(&kGlass);
	PaintRect(&r);
	TextFont(kGeneva);								/* the maker's lettering		*/
	TextSize(10);
	TextFace(bold);
	TextMode(srcOr);
	RGBForeColor(&kLabel);
	MoveTo(16, 20);
	DrawString("\pM A C - 5 9");

	SetRect(&r, 6, 74, 290, 110);					/* the program-card strip		*/
	RGBForeColor(&kCard);
	PaintRect(&r);
	RGBForeColor(&kCardLn);
	MoveTo(6, 92);  LineTo(289, 92);
	MoveTo(6, 109); LineTo(289, 109);
	for (x = 62; x < 290; x += 56) { MoveTo(x, 92); LineTo(x, 109); }
	for (x = 0; x < 5; x++)							/* the cue card, one cell per key A-E	*/
		if (cue[x][0] > 0) {
			w = StringWidth(cue[x]);
			MoveTo((short) (38 + 56 * x - w / 2), 105);
			DrawString(cue[x]);
		}
	TextSize(9);
	RGBForeColor(&kCardTx);
	MoveTo(12, 87);
	if (cardName[0] > 0) DrawString(cardName);
	else DrawString("\pMAGNETKARTE");

	SetRect(&r, 6, 114, 290, 542);					/* the key panel				*/
	RGBForeColor(&kPanel);
	PaintRect(&r);
	InsetRect(&r, 3, 3);
	RGBForeColor(&kSilver);
	FrameRect(&r);

	TextFace(normal);								/* the model name, as printed	*/
	TextSize(12);									/* under the keys				*/
	RGBForeColor(&kLabel);
	w = StringWidth("\pProgrammable 59");
	MoveTo((short) (282 - w), 524);
	DrawString("\pProgrammable 59");
	TextSize(9);
	w = StringWidth("\pSolid State Software");
	MoveTo((short) (282 - w), 535);
	DrawString("\pSolid State Software");

	MoveTo(14, 535);								/* status: 2nd INV Deg Fix n	*/
	DrawString(info);

	SetRect(&r, 300, 12, 492, 538);					/* the printer strip			*/
	RGBForeColor(&kPaper);
	PaintRect(&r);

	RGBForeColor(&kBlackC);
	RGBBackColor(&kWhiteC);
}


/******************************************************************************
 CKeyPane

	The pane is 48 x 40: the 2nd label in the top 14 pixels, the key in the
	26 below. The key changes colour while pressed and sends its command
	when the mouse is released over it.

 ******************************************************************************/

CKeyPane::CKeyPane(CView *anEnclosure, CBureaucrat *aSupervisor,
				   short aWidth, short aHeight, short aHEncl, short aVEncl,
				   const char *aLegend, const char *aSecond, short aStyle, long aCommand)
	: CPane(anEnclosure, aSupervisor, aWidth, aHeight, aHEncl, aVEncl,
			sizFIXEDLEFT, sizFIXEDTOP)
{
	CToPStr(aLegend, legend);
	CToPStr(aSecond, second);
	style = aStyle;
	command = aCommand;
	latched = FALSE;
	helpID = 0;
	SetWantsClicks(TRUE);
}


void CKeyPane::SetHelp(short anIndex, short aResID)
{
	helpResIndex = anIndex;
	helpID = aResID;
}

short CKeyPane::GetHelpResID()
{
	if (helpID == kKeyHelpID && gTI59 != NULL && gTI59->SecondPending()) return kKey2ndHelpID;
	return helpID != 0 ? helpID : CPane::GetHelpResID();
}


/* A latching key (the printer's TRACE) stays down until pressed again. */
void CKeyPane::SetLatched(Boolean aLatched)
{
	Rect r;

	latched = aLatched;
	Prepare();
	KeyRect(&r);
	DrawKey(&r, latched);
}


void CKeyPane::KeyRect(Rect *r)
{
	FrameToQDR(&frame, r);
	r->top += 14;
}


void CKeyPane::Draw(Rect *area)
{
	Rect	r, k;
	short	w, x;

	FrameToQDR(&frame, &r);
	RGBForeColor(&kPanel);
	PaintRect(&r);
	if (second[0] > 0) {
		TextFont(kGeneva);
		TextSize(9);
		TextFace(normal);
		TextMode(srcOr);
		RGBForeColor(&kLabel);
		if (PEqual(second, "xbar")) {					/* x with a bar over it		*/
			w = CharWidth('x');
			x = (short) ((r.left + r.right - w) / 2);
			MoveTo(x, (short) (r.top + 11));
			DrawChar('x');
			MoveTo(x, (short) (r.top + 3));
			LineTo((short) (x + w - 1), (short) (r.top + 3));
		}
		else {
			w = StringWidth(second);
			MoveTo((short) ((r.left + r.right - w) / 2), (short) (r.top + 11));
			DrawString(second);
		}
	}
	KeyRect(&k);
	DrawKey(&k, latched);
}


void CKeyPane::DrawKey(Rect *r, Boolean pressed)
{
	RGBColor	face;
	Rect		k = *r;
	short		w, i, cx, cy, size;
	Str255		base, sup;

	RGBForeColor(&kPanel);
	PaintRect(&k);
	InsetRect(&k, 2, 1);
	LightenFor(style, pressed, &face);
	if (pressed && latched && style == kKeySecond) face = kSecondOn;	/* 2nd waits for its key: lit	*/
	RGBForeColor(&kShadow);								/* the key's shadow			*/
	OffsetRect(&k, 1, 2);
	PaintRoundRect(&k, 6, 6);
	OffsetRect(&k, -1, -2);
	RGBForeColor(&face);
	PaintRoundRect(&k, 6, 6);
	if (!pressed) {
		RGBForeColor(style == kKeyDark ? &kHiLite : &kWhiteC);
		MoveTo((short) (k.left + 3), (short) (k.top + 1));
		LineTo((short) (k.right - 4), (short) (k.top + 1));
	}

	TextFont(kGeneva);
	TextFace(bold);
	TextMode(srcOr);
	size = (legend[0] == 1) ? 12 : 10;
	TextSize(size);
	RGBForeColor(style == kKeyDark ? &kDarkTx : style == kKeySecond ? &kSecondT : &kBlackC);
	cx = (short) ((k.left + k.right) / 2);
	cy = (short) (k.bottom - 8);

	if (PEqual(legend, "*")) {							/* the multiplication cross	*/
		PenSize(2, 2);
		MoveTo((short) (cx - 4), (short) (cy - 9)); LineTo((short) (cx + 3), (short) (cy - 2));
		MoveTo((short) (cx + 3), (short) (cy - 9)); LineTo((short) (cx - 4), (short) (cy - 2));
		PenSize(1, 1);
	}
	else if (PEqual(legend, "-")) {						/* a long minus				*/
		PenSize(2, 2);
		MoveTo((short) (cx - 5), (short) (cy - 6)); LineTo((short) (cx + 4), (short) (cy - 6));
		PenSize(1, 1);
	}
	else {
		for (i = 1; i <= legend[0] && legend[i] != '^'; i++) ;
		if (i <= legend[0]) {							/* "x^2", "y^x": superscript	*/
			BlockMove(legend + 1, base + 1, (long) (i - 1)); base[0] = (unsigned char) (i - 1);
			BlockMove(legend + i + 1, sup + 1, (long) (legend[0] - i)); sup[0] = (unsigned char) (legend[0] - i);
			TextSize(12);
			w = StringWidth(base);
			TextSize(9);
			w += StringWidth(sup);
			TextSize(12);
			MoveTo((short) (cx - w / 2), cy);
			DrawString(base);
			TextSize(9);
			Move(0, -5);
			DrawString(sup);
		}
		else {
			w = StringWidth(legend);
			MoveTo((short) (cx - w / 2), cy);
			DrawString(legend);
		}
	}

	TextFace(normal);
	RGBForeColor(&kBlackC);
	RGBBackColor(&kWhiteC);
}


void CKeyPane::DoClick(Point hitPt, short modifierKeys, long when)
{
	Rect	r;
	Point	pt;
	Boolean	in, was;

	Prepare();
	KeyRect(&r);
	in = PtInRect(hitPt, &r);
	if (!in) return;								/* a click on the 2nd label		*/
	DrawKey(&r, TRUE);
	while (StillDown()) {
		GetMouse(&pt);
		was = in;
		in = PtInRect(pt, &r);
		if (in != was) DrawKey(&r, in);
	}
	DrawKey(&r, latched);
	if (in && itsSupervisor != NULL)
		itsSupervisor->DoCommand(command);
}
