/******************************************************************************
 CMain.c

				CMain Document Class

	Generated by Visual Architect - extended by Pit Förster with Claude Code via AppleBridge
	(2026-10-26): "Rechner 2", the TI-59 from its ROM.

	Rechner's TI-59 case and keys around the TMC0501 emulator of Calc-U-59
	(TI59Core.cp, GPL-3.0, (c) 2026 Martin Erzberger). Nothing is computed
	here: a key presses the emulated keyboard matrix (row*10 + col), the ROM
	does the rest, and the display shows what the ROM multiplexes onto the
	twelve LEDs, drawn as seven-segment cells. The PC-100C printer gets a
	window of its own next to the calculator.

	Timing: an idle chore runs the CPU at the TI-59's clock (455 kHz / 32 =
	14 219 instructions per second, an IDLE instruction counts four), or as
	fast as the Mac can if "Volle Geschwindigkeit" is on. A key is held for
	0.1 s of TI time and released; the next waits until the keyboard loop
	runs again (or 0.4 s, so R/S still reaches a running program).

	The ROM (Texas Instruments' firmware, private use) is resource 'TROM' 128
	in Project Resources.rsrc: 6144 words big-endian + 1024 constant nibbles.

	Layout: window 296 x 548 - case 0,0; display 12,26 272 x 38; keys 14,118
	in 5 columns x 9 rows, pitch 56 x 44. Printer window 208 x 548 beside it.

	The source is plain ASCII; characters outside it are MacRoman escapes.

 ******************************************************************************/

	// Mac keyboard -> matrix code (0: not a calculator key).
static short KeyForChar(char c)
{
	switch (c) {
		case '0': return 92;	case '1': return 82;	case '2': return 83;	case '3': return 84;
		case '4': return 72;	case '5': return 73;	case '6': return 74;
		case '7': return 62;	case '8': return 63;	case '9': return 64;
		case '.': case ',': return 93;
		case '+': return 85;	case '-': return 75;	case '*': return 65;	case '/': return 55;
		case '=': case 0x0D: case 0x03: return 95;
		case '(': return 53;	case ')': return 54;	case '^': return 45;
		case 'e': return 52;						/* EE						*/
		case 0x1B: return 25;						/* Escape: CLR				*/
		case 0x08: return 24;						/* Delete: CE				*/
		case 'A': return 11;	case 'B': return 12;	case 'C': return 13;
		case 'D': return 14;	case 'E': return 15;
		case 'n': return 21;						/* 2nd						*/
		case 'i': return 22;						/* INV						*/
		case 'l': return 31;						/* LRN						*/
		case 'r': return 91;						/* R/S						*/
	}
	return 0;
}
