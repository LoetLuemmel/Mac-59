static void CToPStr(const char *c, Str255 p)
{
	short n = 0;
	while (c[n] != '\0' && n < 255) { p[n + 1] = (unsigned char) c[n]; n++; }
	p[0] = (unsigned char) n;
}

static short CLen(const char *c)
{
	short n = 0;
	while (c[n] != '\0') n++;
	return n;
}

static void CCat(char *dst, const char *src)
{
	short n = CLen(dst), i = 0;
	while (src[i] != '\0') dst[n++] = src[i++];
	dst[n] = '\0';
}

static void CCopy(char *dst, const char *src)
{
	dst[0] = '\0';
	CCat(dst, src);
}

static Boolean PEqual(Str255 p, const char *c)
{
	short i, n = CLen(c);
	if (p[0] != n) return FALSE;
	for (i = 0; i < n; i++) if (p[i + 1] != (unsigned char) c[i]) return FALSE;
	return TRUE;
}
